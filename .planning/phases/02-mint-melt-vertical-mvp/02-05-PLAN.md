    ```python
    import pytest
    from lnurlmint.crud import settle_mint, get_note, db

    @pytest.mark.asyncio
    async def test_poc_a2_settle_race(node, db_setup):
        """TEST-02: Compare-and-set UPDATE ... WHERE minted=0 + INSERT is atomic.

        Two concurrent settle_mint calls for the same payment_hash result in
        exactly one note — the compare-and-set ensures only the winner inserts.
        """
        # Insert a pending mint record
        payment_hash = "deadbeef" * 8
        async with db.connect() as conn:
            await conn.execute(
                "INSERT INTO lnurlmint.mints_records "
                "(payment_hash, mint_id, pr, amount_msat, minted) "
                "VALUES (:ph, :mid, :pr, :amount, 0)",
                {"ph": payment_hash, "mid": "testmint", "pr": "lnbc...", "amount": 5000}
            )

        # First settle — winner
        result1 = await settle_mint(payment_hash)
        assert result1 == 5000, f"first settle should return amount, got {result1}"

        # Second settle — loser (already settled)
        result2 = await settle_mint(payment_hash)
        assert result2 is None, "second settle should return None (already settled)"

        # Exactly one note exists
        note = await get_note(payment_hash, "testmint")
        assert note is not None, "note should be materialized"
        assert note.amount_msat == 5000

        # Cleanup
        async with db.connect() as conn:
            await conn.execute("DELETE FROM lnurlmint.notes WHERE id = :id", {"id": payment_hash})
            await conn.execute("DELETE FROM lnurlmint.mints_records WHERE payment_hash = :ph", {"ph": payment_hash})
    ```

    **Key assertion**: The second `settle_mint` call returns `None` because `UPDATE ... WHERE minted=0` affects 0 rows (minted is already 1 from the first call). The `rowcount != 1` check prevents the loser from inserting a duplicate note.
